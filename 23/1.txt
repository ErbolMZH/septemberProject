},

  isCreditCategory() {
    // Укажите код категории Кредит из вашего справочника (обычно 'B' или 'CREDIT')
    return this.form.transaction_category === 'B' || this.form.transaction_category === 'CREDIT';
  }
},

methods: {
  handleSuspendFiles(event) {
    this.form.suspend_files = Array.from(event.target.files);
  },
  handleForgiveFiles(event) {
    this.form.forgive_files = Array.from(event.target.files);
  }
}
