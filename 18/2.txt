package handlers

import (
	"net/http"

	pg "af-back/db" // Подключаем твой пакет базы данных

	"github.com/labstack/echo/v4"
)

// RefResponse — DTO для ответа фронтенду
type RefResponse struct {
	Code   string `json:"code"`
	NameRU string `json:"name_ru"`
}

// GetRefsByScheme — простая функция-хэндлер
func GetRefsByScheme(c echo.Context) error {
	refType := c.QueryParam("type")
	if refType == "" {
		refType = "FRAUD_SCHEME"
	}

	query := `
		SELECT code, name_ru 
		FROM refs 
		WHERE type = $1
		ORDER BY id ASC
	`

	// Вызываем QueryContext напрямую через твоё подключение pg.DB (или pg.GetDB())
	rows, err := pg.DB.QueryContext(c.Request().Context(), query, refType)
	if err != nil {
		return c.JSON(http.StatusInternalServerError, map[string]string{
			"error": "Failed to fetch references: " + err.Error(),
		})
	}
	defer rows.Close()

	refs := make([]RefResponse, 0)

	for rows.Next() {
		var item RefResponse
		if err := rows.Scan(&item.Code, &item.NameRU); err != nil {
			return c.JSON(http.StatusInternalServerError, map[string]string{
				"error": "Error scanning row: " + err.Error(),
			})
		}
		refs = append(refs, item)
	}

	if err := rows.Err(); err != nil {
		return c.JSON(http.StatusInternalServerError, map[string]string{
			"error": "Rows iteration error: " + err.Error(),
		})
	}

	return c.JSON(http.StatusOK, refs)
}
