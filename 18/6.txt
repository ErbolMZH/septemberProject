func GetRefsByScheme(c echo.Context) error {
	refType := c.QueryParam("type")
	code1 := c.QueryParam("code1") // Получаем код родительской категории

	if refType == "" {
		refType = "FRAUD_SCHEME"
	}

	query := `
		SELECT code, name_ru 
		FROM refs 
		WHERE type = $1
	`
	args := []interface{}{refType}

	// Если передан code1, добавляем его в условия SQL-запроса
	if code1 != "" {
		query += " AND code1 = $2"
		args = append(args, code1)
	}

	query += " ORDER BY id ASC"

	rows, err := pg.DB.QueryContext(c.Request().Context(), query, args...)
	if err != nil {
		return c.JSON(http.StatusInternalServerError, map[string]string{
			"error": "Failed to fetch references: " + err.Error(),
		})
	}
	defer rows.Close()

	refs := make([]RefResponse, 0)
	for rows.Next() {
		var item RefResponse
		if err := rows.Scan(&item.Code, &item.NameRU); err != nil {
			return c.JSON(http.StatusInternalServerError, map[string]string{
				"error": "Error scanning row: " + err.Error(),
			})
		}
		refs = append(refs, item)
	}

	return c.JSON(http.StatusOK, refs)
}

watch: {
  // Отслеживаем выбор категории транзакции
  'form.transaction_category'(newCategoryCode) {
    // Сбрасываем выбранный тип транзакции при смене категории
    this.form.type = '';
    this.transactionTypes = [];

    // Если категория выбрана (например, 'P'), подтягиваем нужные типы
    if (newCategoryCode) {
      this.fetchTransactionTypes(newCategoryCode);
    }
  }
},

async fetchTransactionTypes(categoryCode) {
    try {
      // Отправляем type=TRANSACTION_TYPE и code1=P (или A, C, CR)
      const response = await axios.get(`/incidents/refs?type=TRANSACTION_TYPE&code1=${categoryCode}`);
      this.transactionTypes = response.data || [];
    } catch (err) {
      console.error('Ошибка загрузки типов транзакций:', err);
      this.transactionTypes = [];
    }
  }
