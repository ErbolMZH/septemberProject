        this.form.sender_subject_type = 'LEGAL';

    this.fetchRequisiteTypes();

        const response = await fetch('/api/refs?type=REQUISITE_TYPE');

        this.requisiteTypes = data;

        console.error('Ошибка при загрузке типов реквизитов:', error);
      }
