async fetchCurrencies() {
  try {
    const response = await axios.get('/incidents/refs?type=CURRENCY');
    
    // 1. Получаем массив из ответа
    const rawData = (response.data && response.data.refs) ? response.data.refs : [];

    // 2. Коды приоритетных валют
    const priorityCodes = ['398', '978', '840', '648'];

    // 3. Создаем копию массива [...rawData] перед сортировкой
    this.currencies = [...rawData].sort((a, b) => {
      const indexA = priorityCodes.indexOf(String(a.code));
      const indexB = priorityCodes.indexOf(String(b.code));

      // Если обе валюты приоритетные — сохраняем их порядок (398 -> 978 -> 840 -> 648)
      if (indexA !== -1 && indexB !== -1) {
        return indexA - indexB;
      }
      // Если только a приоритетная — поднимаем наверх
      if (indexA !== -1) return -1;
      // Если только b приоритетная — поднимаем наверх
      if (indexB !== -1) return 1;

      // Все остальные сортируем по названию или по коду
      const nameA = a.name_ru || '';
      const nameB = b.name_ru || '';
      return nameA.localeCompare(nameB);
    });

    console.log('Загруженные валюты (итоговый массив):', this.currencies);
  } catch (err) {
    console.error('Ошибка загрузки валют:', err);
    this.currencies = [];
  }
}
