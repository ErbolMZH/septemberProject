export default {
  data() {
    return {
      form: {
        sender_subject_type: '',
        sender_role: '',
        sender_organization_name: '',
        sender_registration_date: '',
        sender_bin: '',
        sender_country: '',
        sender_first_name: '',
        sender_last_name: '',
        sender_middle_name: '',
        sender_iin: '',
        sender_birth_date: '',
        sender_gender: '',
        sender_requisite_type: ''
      }
    };
  },
  watch: {
    isCreditCategory: {
      immediate: true,
      handler(isCR) {
        if (isCR) {
          // Только для кредитных категорий устанавливаем ВТБ и блокируем
          this.form.sender_subject_type = 'LEGAL';
          this.form.sender_organization_name = 'ДО АО Банк ВТБ Казахстан';
          this.form.sender_bin = '080940010300';
          this.form.sender_role = 'victim';
        } else {
          // В остальных категориях очищаем форму — пользователь выбирает сам
          this.clearSenderForm();
        }
      }
    }
  },
  methods: {
    onSubjectTypeChange() {
      // Сохраняем выбранный тип и сбрасываем введенные данные при переключении типов
      const selectedType = this.form.sender_subject_type;
      this.clearSenderForm();
      this.form.sender_subject_type = selectedType;
    },
    clearSenderForm() {
      this.form.sender_subject_type = '';
      this.form.sender_role = '';
      this.form.sender_organization_name = '';
      this.form.sender_registration_date = '';
      this.form.sender_bin = '';
      this.form.sender_country = '';
      this.form.sender_first_name = '';
      this.form.sender_last_name = '';
      this.form.sender_middle_name = '';
      this.form.sender_iin = '';
      this.form.sender_birth_date = '';
      this.form.sender_gender = '';
      this.form.sender_requisite_type = '';
    }
  }
};
